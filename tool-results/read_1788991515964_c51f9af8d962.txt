    80→// Relative time in Indonesian — same helper pattern as whatsapp-log.tsx / parent-portal.tsx.
    81→function timeAgo(iso: string): string {
    82→  const diff = Date.now() - new Date(iso).getTime()
    83→  const m = Math.floor(diff / 60000)
    84→  if (m < 1) return 'baru saja'
    85→  if (m < 60) return `${m} menit lalu`
    86→  const h = Math.floor(m / 60)
    87→  if (h < 24) return `${h} jam lalu`
    88→  const d = Math.floor(h / 24)
    89→  if (d < 7) return `${d} hari lalu`
    90→  return formatShortDate(iso)
    91→}

   389→export function GuruOverview({ user, onNavigate }: {

   403→  const load = useCallback(async (quiet = false) => {
   404→    if (!quiet) {
   405→      setLoading(true)
   406→      setError(null)
   407→    }
   408→    try {
   409→      const [classes, sessions, hafalans, students] = await Promise.all([
   410→        apiGet<ClassRoom[]>('/api/classes'),
   411→        apiGet<SessionItem[]>('/api/sessions?active=1'),
   412→        apiGet<Hafalan[]>('/api/hafalan'),
   413→        apiGet<Student[]>('/api/students'),
   414→      ])
   415→      if (!mountedRef.current) return
   416→      setData({ classes, sessions, hafalans, students })
   417→    } catch (e) {
   418→      if (!mountedRef.current) return
   419→      if (quiet) return // refresh senyap: biarkan data lama tampil tanpa Alert penuh
   420→      setError(e instanceof Error ? e.message : 'Gagal memuat data ringkasan')
   421→    } finally {
   422→      if (mountedRef.current && !quiet) setLoading(false)
   423→    }
   424→  }, [])
   425→
   426→  useEffect(() => {
   427→    mountedRef.current = true
   428→    void load()
   429→    return () => {
   430→      mountedRef.current = false
   431→    }
   432→  }, [load])
   433→
   434→  function copyCode(code: string) {
   435→    void navigator.clipboard
   436→      ?.writeText(code)
   437→      .then(() => toast({ title: 'Kode disalin', description: `Kode sesi ${code} siap dibagikan.` }))
   438→      .catch(() => toast({ title: 'Gagal menyalin', description: 'Salin kode secara manual.' }))
   439→  }
   440→
   441→  if (loading) {
   442→    return (
   443→      <div className="space-y-5">
   444→        <Skeleton className="h-44 rounded-2xl" />
   445→        <div className="grid grid-cols-2 gap-3 lg:grid-cols-4 lg:gap-4">
   446→          {Array.from({ length: 4 }).map((_, i) => (
   447→            <Skeleton key={i} className="h-20 rounded-2xl" />
   448→          ))}
   449→        </div>
   450→        <div className="grid gap-4 lg:grid-cols-2">
   451→          <Skeleton className="h-72 rounded-2xl" />
   452→          <Skeleton className="h-72 rounded-2xl" />
   453→        </div>
   454→      </div>
   455→    )
   456→  }
   457→
   458→  if (error) {
   459→    return (
   460→      <Alert variant="destructive" className="rounded-2xl">
   461→        <AlertCircle className="size-4" />
   462→        <AlertTitle>Gagal memuat data ringkasan</AlertTitle>
   463→        <AlertDescription>
   464→          {error}
   465→          <div className="mt-3">
   466→            <Button variant="outline" className="h-11" onClick={() => void load()}>
   467→              <RefreshCw className="size-4" /> Coba Lagi
   468→            </Button>
   469→          </div>
   470→        </AlertDescription>
   471→      </Alert>
   472→    )
   473→  }
   474→
   475→  if (!data) return null
   476→
   477→  // ==== Client-side ownership filtering ====
   478→  const ownClasses = data.classes.filter((c) => c.teacherId && c.teacherId === user.teacherId)
   479→  const ownClassIds = new Set(ownClasses.map((c) => c.id))
   480→  const ownSessions = data.sessions.filter((s) => ownClassIds.has(s.classId))
   481→  const ownStudents = data.students.filter((st) => st.classId && ownClassIds.has(st.classId))
   482→  const ownStudentCount = ownStudents.filter((st) => st.status === 'AKTIF').length
   483→  const ownStudentIds = new Set(ownStudents.map((st) => st.id))
   484→  const ownHafalan = data.hafalans.filter((h) => ownStudentIds.has(h.studentId))
   485→  const hafalanCapped = data.hafalans.length >= 100
   486→  const recentHafalan = ownHafalan.slice(0, 8)
   487→
   488→  // ==== Progres target santri (Task 13-a) — murni dari state yang sudah dimuat ====
   489→  const hafalanByStudent = new Map<string, Hafalan[]>()
   490→  for (const h of ownHafalan) {
   491→    const list = hafalanByStudent.get(h.studentId)
   492→    if (list) list.push(h)
   493→    else hafalanByStudent.set(h.studentId, [h])
   494→  }
   495→  // Satu baris per santri kelas sendiri yang PUNYA target valid di peta Juz 30.
   496→  const targetRows = ownStudents
   497→    .flatMap((st) => {
   498→      const tp = targetProgress(hafalanByStudent.get(st.id) ?? [], st.hafalanTarget)
