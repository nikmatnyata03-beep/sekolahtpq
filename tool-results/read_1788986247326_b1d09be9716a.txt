    85→type FieldError = Partial<Record<'childName' | 'birthDate' | 'parentName' | 'phone' | 'email' | 'address', string>>

    98→function validate(form: FormState): FieldError {
    99→  const errors: FieldError = {}
   100→  if (!form.childName.trim()) errors.childName = 'Nama lengkap anak wajib diisi.'
   101→  else if (form.childName.trim().length < 3) errors.childName = 'Nama minimal 3 karakter.'
   102→  if (!form.birthDate) errors.birthDate = 'Tanggal lahir wajib diisi.'
   103→  if (!form.parentName.trim()) errors.parentName = 'Nama orang tua/wali wajib diisi.'
   104→  if (!form.phone.trim()) errors.phone = 'Nomor WhatsApp wajib diisi.'
   105→  else if (!/^[0-9+\-\s]{9,16}$/.test(form.phone.trim())) errors.phone = 'Nomor WhatsApp tidak valid (9–16 digit).'
   106→  if (form.email.trim() && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())) errors.email = 'Format email tidak valid.'
   107→  if (!form.address.trim()) errors.address = 'Alamat domisili wajib diisi.'
   108→  return errors
   109→}
   110→
   111→function ErrorText({ message }: { message?: string }) {
   112→  if (!message) return null
   113→  return (
   114→    <p className="flex items-center gap-1 text-xs text-red-600" role="alert">
   115→      <AlertCircle className="size-3 shrink-0" />
   116→      {message}
   117→    </p>
   118→  )
   119→}

   359→                  onChange={(e) => setField('email', e.target.value)}

   361→                />
   362→                <ErrorText message={errors.email} />
   363→              </div>
   364→              <div className="space-y-1.5 md:col-span-2">

   366→                  Alamat Domisili <span className="text-red-600">*</span>

   370→                  rows={2}
   371→                  placeholder="cth. Jl. Merpati Raya No. 25, Cibubur, Jakarta Timur"

   373→                  onChange={(e) => setField('address', e.target.value)}

   376→                <ErrorText message={errors.address} />

   380→            {/* Dokumen */}

   409→                id="ppdb-note"

   417→            <Button

   422→            >
   423→              {submitting ? (
   424→                <>
   425→                  <Loader2 className="size-4 animate-spin" />
   426→                  Mengirim Pendaftaran…
   427→                </>
   428→              ) : (

   431→                  <ArrowRight className="size-4" />
   432→                </>
   433→              )}
   434→            </Button>
   435→          </form>
   436→
   437→          {/* ================= KOLOM KANAN ================= */}
   438→          <div className="flex flex-col gap-6 lg:col-span-2">
   439→            {/* Cek status pendaftaran */}
   440→            <div className="rounded-2xl border border-emerald-100 bg-white p-6 shadow-sm">

   442→                <span className="flex size-9 items-center justify-center rounded-xl bg-amber-50 text-amber-600">
   443→                  <SearchCheck className="size-4.5" />
   444→                </span>
   445→                Cek Status Pendaftaran
   446→              </h3>
