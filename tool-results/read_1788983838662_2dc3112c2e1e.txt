    89→function gradeCircleClass(grade: number | null): string {
    90→  if (grade === null) return 'bg-stone-300 text-stone-600'
    91→  if (grade >= 85) return 'bg-emerald-600 text-white'
    92→  if (grade >= 70) return 'bg-amber-500 text-white'
    93→  return 'bg-red-500 text-white'
    94→}

    96→/** Text-only grade coloring used inside the printable report table. */
    97→function gradeTextClass(grade: number | null): string {
    98→  if (grade === null) return 'text-stone-400'
    99→  if (grade >= 85) return 'text-emerald-700'
   100→  if (grade >= 70) return 'text-amber-600'
   101→  return 'text-red-600'
   102→}

   104→function attendanceRate(child: ParentStudent): number {
   105→  const { hadir, total } = child.attendanceSummary
   106→  return total > 0 ? Math.round((hadir / total) * 100) : 0
   107→}
   108→
   109→function averageHafalanGrade(hafalans: Hafalan[]): number | null {
   110→  const grades = hafalans.map((h) => h.grade).filter((g): g is number => typeof g === 'number')
   111→  if (grades.length === 0) return null
   112→  return Math.round(grades.reduce((a, b) => a + b, 0) / grades.length)
   113→}

   126→function PaymentDialog({
   127→  payment,
   128→  studentName,
   129→  onDismiss,
   130→  onPaid,
   131→}: {
   132→  payment: Payment | null
   133→  studentName: string
   134→  onDismiss: () => void
   135→  onPaid: () => void
   136→}) {
   137→  const [method, setMethod] = useState<PayMethod | null>(null)
   138→  const [step, setStep] = useState<PayStep>('form')
   139→  const [error, setError] = useState<string | null>(null)
   140→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   141→
   142→  useEffect(() => {
   143→    return () => {
   144→      if (timerRef.current) clearTimeout(timerRef.current)
   145→    }
   146→  }, [])
   147→
   148→  function pay() {
   149→    if (!payment || !method) return
   150→    setError(null)
   151→    setStep('processing')
   152→    if (timerRef.current) clearTimeout(timerRef.current)
   153→    // Simulated gateway round-trip (~1.8s) before the "webhook" hits the API
   154→    timerRef.current = setTimeout(() => {
   155→      apiSend('/api/payments', 'PUT', { id: payment.id, status: 'SUCCESS', method })
   156→        .then(() => {
   157→          setStep('success')
   158→          onPaid()
   159→        })
   160→        .catch((err: unknown) => {
   161→          setStep('form')
   162→          setError(err instanceof Error ? err.message : 'Pembayaran gagal. Silakan coba lagi.')
   163→        })
   164→    }, 1800)
   165→  }
   166→
   167→  function handleClose(open: boolean) {
   168→    if (!open && step !== 'processing') onDismiss()
   169→  }
   170→
   171→  const methodLabel = PAY_METHODS.find((m) => m.value === method)?.label ?? method
   172→
   173→  return (
   174→    <Dialog open={!!payment} onOpenChange={handleClose}>
   175→      <DialogContent className="rounded-2xl sm:max-w-md">
   176→        <DialogHeader>
   177→          <DialogTitle>Simulasi Pembayaran</DialogTitle>
   178→          <DialogDescription>Gateway pembayaran demo — bukan transaksi sungguhan.</DialogDescription>
   179→        </DialogHeader>
   180→
   181→        {payment && (
   182→          <>
   183→            {/* Invoice summary */}
   184→            <div className="rounded-xl border border-stone-200 bg-stone-50 p-4">
   185→              <div className="flex items-start justify-between gap-3">
   186→                <div className="min-w-0">
   187→                  <p className="font-mono text-[11px] text-stone-500">{payment.invoiceNo}</p>
   188→                  <p className="mt-0.5 truncate text-sm font-semibold text-stone-800">{payment.title}</p>
   189→                  <p className="text-xs text-stone-500">Untuk: {studentName}</p>
   190→                </div>
   191→                <p className="shrink-0 text-lg font-bold text-emerald-700">{formatRupiah(payment.amount)}</p>
   192→              </div>
   193→            </div>
   194→
   195→            {step === 'form' && (
   196→              <div className="space-y-4">
   197→                {error && (
   198→                  <div className="rounded-xl border border-red-200 bg-red-50 p-3 text-sm text-red-700">{error}</div>
   199→                )}
   200→                <div className="space-y-1.5">
   201→                  <Label>Pilih metode pembayaran</Label>
   202→                  <RadioGroup
   203→                    value={method ?? ''}
   204→                    onValueChange={(v) => setMethod(v as PayMethod)}
   205→                    className="gap-2"
   206→                  >
   207→                    {PAY_METHODS.map((m) => (
   208→                      <label
   209→                        key={m.value}
   210→                        htmlFor={`pay-${m.value}`}
   211→                        className="flex cursor-pointer items-center gap-3 rounded-xl border border-stone-200 bg-white p-3 transition-colors hover:border-emerald-300 [&:has([data-state=checked])]:border-emerald-600 [&:has([data-state=checked])]:bg-emerald-50"
   212→                      >
   213→                        <RadioGroupItem id={`pay-${m.value}`} value={m.value} />
   214→                        <span className="grid size-9 shrink-0 place-items-center rounded-lg bg-emerald-100 text-emerald-700">
   215→                          <m.icon className="size-4" />
   216→                        </span>
   217→                        <span className="min-w-0">
   218→                          <span className="block text-sm font-semibold text-stone-800">{m.label}</span>
   219→                          <span className="block truncate text-xs text-stone-500">{m.desc}</span>
   220→                        </span>
   221→                      </label>
   222→                    ))}
   223→                  </RadioGroup>
   224→                </div>
   225→                <Button
   226→                  onClick={pay}
   227→                  disabled={!method}
   228→                  className="h-10 w-full rounded-xl bg-emerald-700 text-white hover:bg-emerald-800"
   229→                >
   230→                  Bayar Sekarang
   231→                </Button>
   232→                <p className="text-center text-[11px] text-stone-400">
   233→                  Simulasi pembayaran — tidak ada dana sungguhan yang dipotong.
   234→                </p>
   235→              </div>
   236→            )}
   237→
   238→            {step === 'processing' && (
   239→              <div className="flex flex-col items-center gap-3 py-6 text-center">
   240→                <Loader2 className="size-10 animate-spin text-emerald-700" />
   241→                <div>
   242→                  <p className="font-semibold text-stone-800">Menghubungi gateway pembayaran...</p>
   243→                  <p className="mt-1 text-sm text-stone-500">
   244→                    Memproses {formatRupiah(payment.amount)} via {methodLabel}. Mohon jangan tutup dialog ini.
   245→                  </p>
   246→                </div>
   247→                <div className="mt-1 h-1.5 w-40 overflow-hidden rounded-full bg-stone-200">
   248→                  <div className="h-full w-1/2 animate-pulse rounded-full bg-amber-500" />
   249→                </div>
   250→              </div>
   251→            )}
   252→
   253→            {step === 'success' && (
   254→              <div className="flex flex-col items-center gap-3 py-4 text-center">
   255→                <span className="grid size-14 place-items-center rounded-full bg-emerald-100">
   256→                  <CircleCheck className="size-8 text-emerald-600" />
   257→                </span>
   258→                <div>
   259→                  <p className="text-lg font-bold text-stone-800">Pembayaran Berhasil!</p>
   260→                  <p className="mt-0.5 text-sm text-stone-500">
   261→                    {formatRupiah(payment.amount)} · {methodLabel} · {payment.invoiceNo}
   262→                  </p>
   263→                </div>
   264→                <div className="flex items-center gap-2 rounded-xl border border-emerald-200 bg-emerald-50 px-3 py-2 text-xs text-emerald-800">
   265→                  <CheckCircle2 className="size-4 shrink-0" />
   266→                  WhatsApp konfirmasi terkirim ke wali.
   267→                </div>
   268→                <Button
   269→                  onClick={onDismiss}
   270→                  className="mt-1 h-9 w-full rounded-xl bg-emerald-700 text-white hover:bg-emerald-800"
   271→                >
   272→                  Selesai
   273→                </Button>
   274→              </div>
   275→            )}
   276→          </>
   277→        )}
   278→      </DialogContent>
   279→    </Dialog>
   280→  )

   552→
   553→function QrCheckinCard({
   554→  child,
   555→  activeSessions,
   556→  onRefresh,
   557→}: {
   558→  child: ParentStudent
   559→  activeSessions: SessionItem[]
   560→  onRefresh: () => void
   561→}) {
   562→  const { toast } = useToast()
   563→  const [code, setCode] = useState('')
   564→  const [checking, setChecking] = useState(false)
   565→
   566→  // Portal data has no classId, so match the child's class by class name.
   567→  const classSessions = activeSessions.filter((s) => s.className === child.className)
   568→
   569→  async function submit() {
   570→    const trimmed = code.trim()
   571→    if (!trimmed) {
   572→      toast({ title: 'Kode kosong', description: 'Masukkan kode kehadiran terlebih dahulu.', variant: 'destructive' })
   573→      return
   574→    }
   575→    setChecking(true)
   576→    try {
   577→      const res = await apiSend<{ success?: boolean; already?: boolean; message?: string }>(
   578→        '/api/attendance/checkin',
   579→        'POST',
   580→        { code: trimmed, studentId: child.id },
   581→      )
   582→      toast({ title: 'Check-in tercatat', description: res.message ?? `${child.fullName} tercatat hadir.` })
   583→      setCode('')
   584→      onRefresh()
   585→    } catch (err) {
   586→      toast({
   587→        title: 'Check-in gagal',
   588→        description: err instanceof Error ? err.message : 'Terjadi kesalahan saat check-in.',
   589→        variant: 'destructive',
   590→      })
   591→    } finally {
   592→      setChecking(false)
   593→    }
   594→  }
   595→
   596→  return (
   597→    <Card className="rounded-2xl border-emerald-200/70 lg:col-span-2">
   598→      <CardHeader>
   599→        <CardTitle className="flex items-center gap-2 text-base">
   600→          <span className="grid size-8 place-items-center rounded-lg bg-emerald-100 text-emerald-700">
   601→            <ScanLine className="size-4" />
   602→          </span>
   603→          Check-in QR Kehadiran
   604→        </CardTitle>
   605→        <CardDescription>
   606→          Masukkan kode kehadiran dari ustadz/ustadzah untuk mencatat kehadiran {child.fullName} secara mandiri.
   607→        </CardDescription>
   608→      </CardHeader>
   609→      <CardContent className="space-y-3">
   610→        <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   611→          <Input
   612→            value={code}
   613→            onChange={(e) => setCode(e.target.value)}
   614→            onKeyDown={(e) => {
   615→              if (e.key === 'Enter' && !checking) void submit()
   616→            }}
   617→            placeholder="Contoh: DJ-IA01-XXXX"
   618→            className="h-10 flex-1 rounded-xl font-mono uppercase tracking-wider"
   619→            disabled={checking}
   620→          />
   621→          <Button
   622→            onClick={() => void submit()}
   623→            disabled={checking}
   624→            className="h-10 rounded-xl bg-emerald-700 px-5 text-white hover:bg-emerald-800"
   625→          >
   626→            {checking ? (
   627→              <>
   628→                <Loader2 className="size-4 animate-spin" /> Memproses...
   629→              </>
   630→            ) : (
   631→              'Check-in'
   632→            )}
   633→          </Button>
   634→        </div>
   635→        <div className="rounded-xl bg-stone-50 p-3">
   636→          {classSessions.length > 0 ? (
   637→            <div className="flex flex-wrap items-center gap-2">
   638→              <span className="text-xs font-medium text-stone-500">Kode sesi aktif kelas {child.className}:</span>
   639→              {classSessions.map((s) => (
   640→                <button
   641→                  key={s.id}
   642→                  type="button"
   643→                  onClick={() => setCode(s.code)}
   644→                  className="rounded-full border border-amber-300 bg-amber-50 px-2.5 py-1 font-mono text-xs font-semibold text-amber-800 transition-colors hover:bg-amber-100"
   645→                  title={s.topic ? `Topik: ${s.topic}` : 'Sesi aktif'}
   646→                >
   647→                  {s.code}
   648→                </button>
   649→              ))}
   650→            </div>
   651→          ) : (
   652→            <p className="text-xs text-stone-500">
   653→              Tidak ada sesi aktif untuk kelas {child.className} saat ini. Kode muncul di sini saat ustadz/ustadzah
   654→              membuka sesi absensi.
   655→            </p>
   656→          )}
   657→        </div>
   658→      </CardContent>
   659→    </Card>
   660→  )
   661→}
