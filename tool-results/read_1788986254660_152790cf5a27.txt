    98→    98→function validate(form: FormState): FieldError {
    99→    99→  const errors: FieldError = {}
   100→   100→  if (!form.childName.trim()) errors.childName = 'Nama lengkap anak wajib diisi.'
   101→   101→  else if (form.childName.trim().length < 3) errors.childName = 'Nama minimal 3 karakter.'
   102→   102→  if (!form.birthDate) errors.birthDate = 'Tanggal lahir wajib diisi.'
   103→   103→  if (!form.parentName.trim()) errors.parentName = 'Nama orang tua/wali wajib diisi.'
   104→   104→  if (!form.phone.trim()) errors.phone = 'Nomor WhatsApp wajib diisi.'
   105→   105→  else if (!/^[0-9+\-\s]{9,16}$/.test(form.phone.trim())) errors.phone = 'Nomor WhatsApp tidak valid (9–16 digit).'
   106→   106→  if (form.email.trim() && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())) errors.email = 'Format email tidak valid.'
   107→   107→  if (!form.address.trim()) errors.address = 'Alamat domisili wajib diisi.'
   108→   108→  return errors
   109→   109→}

   117→   117→    </p>
   118→   118→  )
   119→   119→}
