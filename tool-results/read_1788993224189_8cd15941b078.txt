   111→   111→function averageHafalanGrade(hafalans: Hafalan[]): number | null {
   112→   112→  const grades = hafalans.map((h) => h.grade).filter((g): g is number => typeof g === 'number')

   114→   114→  return Math.round(grades.reduce((a, b) => a + b, 0) / grades.length)

   556→   556→  child,

   558→   558→  onRefresh,
   559→   559→}: {
   560→   560→  child: ParentStudent
   561→   561→  activeSessions: SessionItem[]
   562→   562→  onRefresh: () => void
   563→   563→}) {
   564→   564→  const { toast } = useToast()
   565→   565→  const [code, setCode] = useState('')
   566→   566→  const [checking, setChecking] = useState(false)
   567→   567→
   568→   568→  // Portal data has no classId, so match the child's class by class name.
   569→   569→  const classSessions = activeSessions.filter((s) => s.className === child.className)
   570→   570→
   571→   571→  async function submit() {
   572→   572→    const trimmed = code.trim()

   574→   574→      toast({ title: 'Kode kosong', description: 'Masukkan kode kehadiran terlebih dahulu.', variant: 'destructive' })

   590→   590→        description: err instanceof Error ? err.message : 'Terjadi kesalahan saat check-in.',
   591→   591→        variant: 'destructive',
   592→   592→      })

   618→   618→            }}
