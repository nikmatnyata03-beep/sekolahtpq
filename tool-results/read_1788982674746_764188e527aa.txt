    88→    88→function averageHafalanGrade(hafalans: Hafalan[]): number | null {

   264→   264→function QrCheckinCard({
   265→   265→  child,
   266→   266→  activeSessions,
   267→   267→  onRefresh,
   268→   268→}: {
   269→   269→  child: ParentStudent
   270→   270→  activeSessions: SessionItem[]
   271→   271→  onRefresh: () => void
   272→   272→}) {
   273→   273→  const { toast } = useToast()
   274→   274→  const [code, setCode] = useState('')
   275→   275→  const [checking, setChecking] = useState(false)
   276→   276→
   277→   277→  // Portal data has no classId, so match the child's class by class name.
   278→   278→  const classSessions = activeSessions.filter((s) => s.className === child.className)
   279→   279→
   280→   280→  async function submit() {
   281→   281→    const trimmed = code.trim()
   282→   282→    if (!trimmed) {
   283→   283→      toast({ title: 'Kode kosong', description: 'Masukkan kode kehadiran terlebih dahulu.', variant: 'destructive' })
   284→   284→      return
   285→   285→    }
   286→   286→    setChecking(true)
   287→   287→    try {
   288→   288→      const res = await apiSend<{ success?: boolean; already?: boolean; message?: string }>(
   289→   289→        '/api/attendance/checkin',
   290→   290→        'POST',
   291→   291→        { code: trimmed, studentId: child.id },
   292→   292→      )
   293→   293→      toast({ title: 'Check-in tercatat', description: res.message ?? `${child.fullName} tercatat hadir.` })
   294→   294→      setCode('')
   295→   295→      onRefresh()
   296→   296→    } catch (err) {
   297→   297→      toast({
   298→   298→        title: 'Check-in gagal',
   299→   299→        description: err instanceof Error ? err.message : 'Terjadi kesalahan saat check-in.',
   300→   300→        variant: 'destructive',
   301→   301→      })
   302→   302→    } finally {
   303→   303→      setChecking(false)
   304→   304→    }
   305→   305→  }
   306→   306→
   307→   307→  return (
   308→   308→    <Card className="rounded-2xl border-emerald-200/70 lg:col-span-2">
   309→   309→      <CardHeader>
   310→   310→        <CardTitle className="flex items-center gap-2 text-base">
   311→   311→          <span className="grid size-8 place-items-center rounded-lg bg-emerald-100 text-emerald-700">
   312→   312→            <ScanLine className="size-4" />
   313→   313→          </span>
   314→   314→          Check-in QR Kehadiran
   315→   315→        </CardTitle>
   316→   316→        <CardDescription>
   317→   317→          Masukkan kode kehadiran dari ustadz/ustadzah untuk mencatat kehadiran {child.fullName} secara mandiri.
   318→   318→        </CardDescription>
   319→   319→      </CardHeader>
   320→   320→      <CardContent className="space-y-3">
   321→   321→        <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   322→   322→          <Input
   323→   323→            value={code}
   324→   324→            onChange={(e) => setCode(e.target.value)}
   325→   325→            onKeyDown={(e) => {
   326→   326→              if (e.key === 'Enter' && !checking) void submit()
   327→   327→            }}
   328→   328→            placeholder="Contoh: DJ-IA01-XXXX"
   329→   329→            className="h-10 flex-1 rounded-xl font-mono uppercase tracking-wider"
   330→   330→            disabled={checking}
   331→   331→          />
   332→   332→          <Button
   333→   333→            onClick={() => void submit()}
   334→   334→            disabled={checking}
   335→   335→            className="h-10 rounded-xl bg-emerald-700 px-5 text-white hover:bg-emerald-800"
   336→   336→          >
   337→   337→            {checking ? (
   338→   338→              <>
   339→   339→                <Loader2 className="size-4 animate-spin" /> Memproses...
   340→   340→              </>
   341→   341→            ) : (
   342→   342→              'Check-in'
   343→   343→            )}
   344→   344→          </Button>
   345→   345→        </div>
   346→   346→        <div className="rounded-xl bg-stone-50 p-3">
   347→   347→          {classSessions.length > 0 ? (
   348→   348→            <div className="flex flex-wrap items-center gap-2">
   349→   349→              <span className="text-xs font-medium text-stone-500">Kode sesi aktif kelas {child.className}:</span>
   350→   350→              {classSessions.map((s) => (
   351→   351→                <button
   352→   352→                  key={s.id}
   353→   353→                  type="button"
   354→   354→                  onClick={() => setCode(s.code)}
   355→   355→                  className="rounded-full border border-amber-300 bg-amber-50 px-2.5 py-1 font-mono text-xs font-semibold text-amber-800 transition-colors hover:bg-amber-100"
   356→   356→                  title={s.topic ? `Topik: ${s.topic}` : 'Sesi aktif'}
   357→   357→                >
   358→   358→                  {s.code}
   359→   359→                </button>
   360→   360→              ))}
   361→   361→            </div>
   362→   362→          ) : (
   363→   363→            <p className="text-xs text-stone-500">
   364→   364→              Tidak ada sesi aktif untuk kelas {child.className} saat ini. Kode muncul di sini saat ustadz/ustadzah
   365→   365→              membuka sesi absensi.
   366→   366→            </p>
   367→   367→          )}
   368→   368→        </div>
