    91→function gradeCircleClass(grade: number | null): string {
    92→  if (grade === null) return 'bg-stone-300 text-stone-600'
    93→  if (grade >= 85) return 'bg-emerald-600 text-white'
    94→  if (grade >= 70) return 'bg-amber-500 text-white'
    95→  return 'bg-red-500 text-white'
    96→}
    97→
    98→/** Text-only grade coloring used inside the printable report table. */
    99→function gradeTextClass(grade: number | null): string {
   100→  if (grade === null) return 'text-stone-400'
   101→  if (grade >= 85) return 'text-emerald-700'
   102→  if (grade >= 70) return 'text-amber-600'
   103→  return 'text-red-600'
   104→}

   106→function attendanceRate(child: ParentStudent): number {
   107→  const { hadir, total } = child.attendanceSummary
   108→  return total > 0 ? Math.round((hadir / total) * 100) : 0
   109→}
   110→
   111→function averageHafalanGrade(hafalans: Hafalan[]): number | null {
   112→  const grades = hafalans.map((h) => h.grade).filter((g): g is number => typeof g === 'number')
   113→  if (grades.length === 0) return null
   114→  return Math.round(grades.reduce((a, b) => a + b, 0) / grades.length)
   115→}

   128→function PaymentDialog({
   129→  payment,
   130→  studentName,
   131→  onDismiss,
   132→  onPaid,
   133→}: {
   134→  payment: Payment | null
   135→  studentName: string
   136→  onDismiss: () => void
   137→  onPaid: () => void
   138→}) {
   139→  const [method, setMethod] = useState<PayMethod | null>(null)
   140→  const [step, setStep] = useState<PayStep>('form')
   141→  const [error, setError] = useState<string | null>(null)
   142→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   143→
   144→  useEffect(() => {
   145→    return () => {
   146→      if (timerRef.current) clearTimeout(timerRef.current)
   147→    }
   148→  }, [])
   149→
   150→  function pay() {
   151→    if (!payment || !method) return
   152→    setError(null)
   153→    setStep('processing')
   154→    if (timerRef.current) clearTimeout(timerRef.current)
   155→    // Simulated gateway round-trip (~1.8s) before the "webhook" hits the API
   156→    timerRef.current = setTimeout(() => {
   157→      apiSend('/api/payments', 'PUT', { id: payment.id, status: 'SUCCESS', method })
   158→        .then(() => {
   159→          setStep('success')
   160→          onPaid()
   161→        })
   162→        .catch((err: unknown) => {
   163→          setStep('form')
   164→          setError(err instanceof Error ? err.message : 'Pembayaran gagal. Silakan coba lagi.')
   165→        })
   166→    }, 1800)
   167→  }
   168→
   169→  function handleClose(open: boolean) {
   170→    if (!open && step !== 'processing') onDismiss()
   171→  }
   172→
   173→  const methodLabel = PAY_METHODS.find((m) => m.value === method)?.label ?? method
   174→
   175→  return (
   176→    <Dialog open={!!payment} onOpenChange={handleClose}>
   177→      <DialogContent className="rounded-2xl sm:max-w-md">
   178→        <DialogHeader>
   179→          <DialogTitle>Simulasi Pembayaran</DialogTitle>
   180→          <DialogDescription>Gateway pembayaran demo — bukan transaksi sungguhan.</DialogDescription>
   181→        </DialogHeader>
   182→
   183→        {payment && (
   184→          <>
   185→            {/* Invoice summary */}
   186→            <div className="rounded-xl border border-stone-200 bg-stone-50 p-4">
   187→              <div className="flex items-start justify-between gap-3">
   188→                <div className="min-w-0">
   189→                  <p className="font-mono text-[11px] text-stone-500">{payment.invoiceNo}</p>
   190→                  <p className="mt-0.5 truncate text-sm font-semibold text-stone-800">{payment.title}</p>
   191→                  <p className="text-xs text-stone-500">Untuk: {studentName}</p>
   192→                </div>
   193→                <p className="shrink-0 text-lg font-bold text-emerald-700">{formatRupiah(payment.amount)}</p>
   194→              </div>
   195→            </div>
   196→
   197→            {step === 'form' && (
   198→              <div className="space-y-4">
   199→                {error && (
   200→                  <div className="rounded-xl border border-red-200 bg-red-50 p-3 text-sm text-red-700">{error}</div>
   201→                )}
   202→                <div className="space-y-1.5">
   203→                  <Label>Pilih metode pembayaran</Label>
   204→                  <RadioGroup
   205→                    value={method ?? ''}
   206→                    onValueChange={(v) => setMethod(v as PayMethod)}
   207→                    className="gap-2"
   208→                  >
   209→                    {PAY_METHODS.map((m) => (
   210→                      <label
   211→                        key={m.value}
   212→                        htmlFor={`pay-${m.value}`}
   213→                        className="flex cursor-pointer items-center gap-3 rounded-xl border border-stone-200 bg-white p-3 transition-colors hover:border-emerald-300 [&:has([data-state=checked])]:border-emerald-600 [&:has([data-state=checked])]:bg-emerald-50"
   214→                      >
   215→                        <RadioGroupItem id={`pay-${m.value}`} value={m.value} />
   216→                        <span className="grid size-9 shrink-0 place-items-center rounded-lg bg-emerald-100 text-emerald-700">
   217→                          <m.icon className="size-4" />
   218→                        </span>
   219→                        <span className="min-w-0">
   220→                          <span className="block text-sm font-semibold text-stone-800">{m.label}</span>
   221→                          <span className="block truncate text-xs text-stone-500">{m.desc}</span>
   222→                        </span>
   223→                      </label>
   224→                    ))}
   225→                  </RadioGroup>
   226→                </div>
   227→                <Button
   228→                  onClick={pay}
   229→                  disabled={!method}
   230→                  className="h-10 w-full rounded-xl bg-emerald-700 text-white hover:bg-emerald-800"
   231→                >
   232→                  Bayar Sekarang
   233→                </Button>
   234→                <p className="text-center text-[11px] text-stone-400">
   235→                  Simulasi pembayaran — tidak ada dana sungguhan yang dipotong.
   236→                </p>
   237→              </div>
   238→            )}
   239→
   240→            {step === 'processing' && (
   241→              <div className="flex flex-col items-center gap-3 py-6 text-center">
   242→                <Loader2 className="size-10 animate-spin text-emerald-700" />
   243→                <div>
   244→                  <p className="font-semibold text-stone-800">Menghubungi gateway pembayaran...</p>
   245→                  <p className="mt-1 text-sm text-stone-500">
   246→                    Memproses {formatRupiah(payment.amount)} via {methodLabel}. Mohon jangan tutup dialog ini.
   247→                  </p>
   248→                </div>
   249→                <div className="mt-1 h-1.5 w-40 overflow-hidden rounded-full bg-stone-200">
   250→                  <div className="h-full w-1/2 animate-pulse rounded-full bg-amber-500" />
   251→                </div>
   252→              </div>
   253→            )}
   254→
   255→            {step === 'success' && (
   256→              <div className="flex flex-col items-center gap-3 py-4 text-center">
   257→                <span className="grid size-14 place-items-center rounded-full bg-emerald-100">
   258→                  <CircleCheck className="size-8 text-emerald-600" />
   259→                </span>
   260→                <div>
   261→                  <p className="text-lg font-bold text-stone-800">Pembayaran Berhasil!</p>
   262→                  <p className="mt-0.5 text-sm text-stone-500">
   263→                    {formatRupiah(payment.amount)} · {methodLabel} · {payment.invoiceNo}
   264→                  </p>
   265→                </div>
   266→                <div className="flex items-center gap-2 rounded-xl border border-emerald-200 bg-emerald-50 px-3 py-2 text-xs text-emerald-800">
   267→                  <CheckCircle2 className="size-4 shrink-0" />
   268→                  WhatsApp konfirmasi terkirim ke wali.
   269→                </div>
   270→                <Button
   271→                  onClick={onDismiss}
   272→                  className="mt-1 h-9 w-full rounded-xl bg-emerald-700 text-white hover:bg-emerald-800"
   273→                >
   274→                  Selesai
   275→                </Button>
   276→              </div>
   277→            )}
   278→          </>
   279→        )}
   280→      </DialogContent>
   281→    </Dialog>
   282→  )

   555→function QrCheckinCard({
   556→  child,
   557→  activeSessions,
   558→  onRefresh,
   559→}: {
   560→  child: ParentStudent
   561→  activeSessions: SessionItem[]
   562→  onRefresh: () => void
   563→}) {
   564→  const { toast } = useToast()
   565→  const [code, setCode] = useState('')
   566→  const [checking, setChecking] = useState(false)
   567→
   568→  // Portal data has no classId, so match the child's class by class name.
   569→  const classSessions = activeSessions.filter((s) => s.className === child.className)
   570→
   571→  async function submit() {
   572→    const trimmed = code.trim()
   573→    if (!trimmed) {
   574→      toast({ title: 'Kode kosong', description: 'Masukkan kode kehadiran terlebih dahulu.', variant: 'destructive' })
   575→      return
   576→    }
   577→    setChecking(true)
   578→    try {
   579→      const res = await apiSend<{ success?: boolean; already?: boolean; message?: string }>(
   580→        '/api/attendance/checkin',
   581→        'POST',
   582→        { code: trimmed, studentId: child.id },
   583→      )
   584→      toast({ title: 'Check-in tercatat', description: res.message ?? `${child.fullName} tercatat hadir.` })
   585→      setCode('')
   586→      onRefresh()
   587→    } catch (err) {
   588→      toast({
   589→        title: 'Check-in gagal',
   590→        description: err instanceof Error ? err.message : 'Terjadi kesalahan saat check-in.',
   591→        variant: 'destructive',
   592→      })
   593→    } finally {
   594→      setChecking(false)
   595→    }
   596→  }
   597→
   598→  return (
   599→    <Card className="rounded-2xl border-emerald-200/70 lg:col-span-2">
   600→      <CardHeader>
   601→        <CardTitle className="flex items-center gap-2 text-base">
   602→          <span className="grid size-8 place-items-center rounded-lg bg-emerald-100 text-emerald-700">
   603→            <ScanLine className="size-4" />
   604→          </span>
   605→          Check-in QR Kehadiran
   606→        </CardTitle>
   607→        <CardDescription>
   608→          Masukkan kode kehadiran dari ustadz/ustadzah untuk mencatat kehadiran {child.fullName} secara mandiri.
   609→        </CardDescription>
   610→      </CardHeader>
   611→      <CardContent className="space-y-3">
   612→        <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   613→          <Input
   614→            value={code}
   615→            onChange={(e) => setCode(e.target.value)}
   616→            onKeyDown={(e) => {
   617→              if (e.key === 'Enter' && !checking) void submit()
   618→            }}
   619→            placeholder="Contoh: DJ-IA01-XXXX"
   620→            className="h-10 flex-1 rounded-xl font-mono uppercase tracking-wider"
   621→            disabled={checking}
   622→          />
   623→          <Button
   624→            onClick={() => void submit()}
   625→            disabled={checking}
   626→            className="h-10 rounded-xl bg-emerald-700 px-5 text-white hover:bg-emerald-800"
   627→          >
   628→            {checking ? (
   629→              <>
   630→                <Loader2 className="size-4 animate-spin" /> Memproses...
   631→              </>
   632→            ) : (
   633→              'Check-in'
   634→            )}
   635→          </Button>
   636→        </div>
   637→        <div className="rounded-xl bg-stone-50 p-3">
   638→          {classSessions.length > 0 ? (
   639→            <div className="flex flex-wrap items-center gap-2">
   640→              <span className="text-xs font-medium text-stone-500">Kode sesi aktif kelas {child.className}:</span>
   641→              {classSessions.map((s) => (
   642→                <button
   643→                  key={s.id}
   644→                  type="button"
   645→                  onClick={() => setCode(s.code)}
   646→                  className="rounded-full border border-amber-300 bg-amber-50 px-2.5 py-1 font-mono text-xs font-semibold text-amber-800 transition-colors hover:bg-amber-100"
   647→                  title={s.topic ? `Topik: ${s.topic}` : 'Sesi aktif'}
   648→                >
   649→                  {s.code}
   650→                </button>
   651→              ))}
   652→            </div>
   653→          ) : (
   654→            <p className="text-xs text-stone-500">
   655→              Tidak ada sesi aktif untuk kelas {child.className} saat ini. Kode muncul di sini saat ustadz/ustadzah
   656→              membuka sesi absensi.
   657→            </p>
   658→          )}
   659→        </div>
   660→      </CardContent>
   661→    </Card>
   662→  )
   663→}
