    76→function gradeCircleClass(grade: number | null): string {
    77→  if (grade === null) return 'bg-stone-300 text-stone-600'
    78→  if (grade >= 85) return 'bg-emerald-600 text-white'
    79→  if (grade >= 70) return 'bg-amber-500 text-white'
    80→  return 'bg-red-500 text-white'
    81→}
    82→
    83→function attendanceRate(child: ParentStudent): number {
    84→  const { hadir, total } = child.attendanceSummary
    85→  return total > 0 ? Math.round((hadir / total) * 100) : 0
    86→}
    87→
    88→function averageHafalanGrade(hafalans: Hafalan[]): number | null {
    89→  const grades = hafalans.map((h) => h.grade).filter((g): g is number => typeof g === 'number')
    90→  if (grades.length === 0) return null
    91→  return Math.round(grades.reduce((a, b) => a + b, 0) / grades.length)
    92→}

    99→const PAY_METHODS: { value: PayMethod; label: string; desc: string; icon: LucideIcon }[] = [
   100→  { value: 'QRIS', label: 'QRIS', desc: 'Scan QR via e-wallet / m-banking', icon: QrCode },

   105→function PaymentDialog({
   106→  payment,
   107→  studentName,
   108→  onDismiss,
   109→  onPaid,
   110→}: {
   111→  payment: Payment | null
   112→  studentName: string
   113→  onDismiss: () => void
   114→  onPaid: () => void
   115→}) {
   116→  const [method, setMethod] = useState<PayMethod | null>(null)
   117→  const [step, setStep] = useState<PayStep>('form')
   118→  const [error, setError] = useState<string | null>(null)
   119→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   120→
   121→  useEffect(() => {
   122→    return () => {
   123→      if (timerRef.current) clearTimeout(timerRef.current)
   124→    }
   125→  }, [])
   126→
   127→  function pay() {
   128→    if (!payment || !method) return
   129→    setError(null)
   130→    setStep('processing')
   131→    if (timerRef.current) clearTimeout(timerRef.current)
   132→    // Simulated gateway round-trip (~1.8s) before the "webhook" hits the API
   133→    timerRef.current = setTimeout(() => {
   134→      apiSend('/api/payments', 'PUT', { id: payment.id, status: 'SUCCESS', method })
   135→        .then(() => {
   136→          setStep('success')
   137→          onPaid()
   138→        })
   139→        .catch((err: unknown) => {
   140→          setStep('form')
   141→          setError(err instanceof Error ? err.message : 'Pembayaran gagal. Silakan coba lagi.')
   142→        })
   143→    }, 1800)
   144→  }
   145→
   146→  function handleClose(open: boolean) {
   147→    if (!open && step !== 'processing') onDismiss()
   148→  }
   149→
   150→  const methodLabel = PAY_METHODS.find((m) => m.value === method)?.label ?? method
   151→
   152→  return (
   153→    <Dialog open={!!payment} onOpenChange={handleClose}>
   154→      <DialogContent className="rounded-2xl sm:max-w-md">
   155→        <DialogHeader>
   156→          <DialogTitle>Simulasi Pembayaran</DialogTitle>
   157→          <DialogDescription>Gateway pembayaran demo — bukan transaksi sungguhan.</DialogDescription>
   158→        </DialogHeader>
   159→
   160→        {payment && (
   161→          <>
   162→            {/* Invoice summary */}
   163→            <div className="rounded-xl border border-stone-200 bg-stone-50 p-4">
   164→              <div className="flex items-start justify-between gap-3">
   165→                <div className="min-w-0">
   166→                  <p className="font-mono text-[11px] text-stone-500">{payment.invoiceNo}</p>
   167→                  <p className="mt-0.5 truncate text-sm font-semibold text-stone-800">{payment.title}</p>
   168→                  <p className="text-xs text-stone-500">Untuk: {studentName}</p>
   169→                </div>
   170→                <p className="shrink-0 text-lg font-bold text-emerald-700">{formatRupiah(payment.amount)}</p>
   171→              </div>
   172→            </div>
   173→
   174→            {step === 'form' && (
   175→              <div className="space-y-4">
   176→                {error && (
   177→                  <div className="rounded-xl border border-red-200 bg-red-50 p-3 text-sm text-red-700">{error}</div>
   178→                )}
   179→                <div className="space-y-1.5">
   180→                  <Label>Pilih metode pembayaran</Label>
   181→                  <RadioGroup
   182→                    value={method ?? ''}
   183→                    onValueChange={(v) => setMethod(v as PayMethod)}
   184→                    className="gap-2"
   185→                  >
   186→                    {PAY_METHODS.map((m) => (
   187→                      <label
   188→                        key={m.value}
   189→                        htmlFor={`pay-${m.value}`}
   190→                        className="flex cursor-pointer items-center gap-3 rounded-xl border border-stone-200 bg-white p-3 transition-colors hover:border-emerald-300 [&:has([data-state=checked])]:border-emerald-600 [&:has([data-state=checked])]:bg-emerald-50"
   191→                      >
   192→                        <RadioGroupItem id={`pay-${m.value}`} value={m.value} />
   193→                        <span className="grid size-9 shrink-0 place-items-center rounded-lg bg-emerald-100 text-emerald-700">
   194→                          <m.icon className="size-4" />
   195→                        </span>
   196→                        <span className="min-w-0">
   197→                          <span className="block text-sm font-semibold text-stone-800">{m.label}</span>
   198→                          <span className="block truncate text-xs text-stone-500">{m.desc}</span>
   199→                        </span>
   200→                      </label>
   201→                    ))}
   202→                  </RadioGroup>
   203→                </div>
   204→                <Button
   205→                  onClick={pay}
   206→                  disabled={!method}
   207→                  className="h-10 w-full rounded-xl bg-emerald-700 text-white hover:bg-emerald-800"
   208→                >
   209→                  Bayar Sekarang
   210→                </Button>
   211→                <p className="text-center text-[11px] text-stone-400">
   212→                  Simulasi pembayaran — tidak ada dana sungguhan yang dipotong.
   213→                </p>
   214→              </div>
   215→            )}
   216→
   217→            {step === 'processing' && (
   218→              <div className="flex flex-col items-center gap-3 py-6 text-center">
   219→                <Loader2 className="size-10 animate-spin text-emerald-700" />
   220→                <div>
   221→                  <p className="font-semibold text-stone-800">Menghubungi gateway pembayaran...</p>
   222→                  <p className="mt-1 text-sm text-stone-500">
   223→                    Memproses {formatRupiah(payment.amount)} via {methodLabel}. Mohon jangan tutup dialog ini.
   224→                  </p>
   225→                </div>
   226→                <div className="mt-1 h-1.5 w-40 overflow-hidden rounded-full bg-stone-200">
   227→                  <div className="h-full w-1/2 animate-pulse rounded-full bg-amber-500" />
   228→                </div>
   229→              </div>
   230→            )}
   231→
   232→            {step === 'success' && (
   233→              <div className="flex flex-col items-center gap-3 py-4 text-center">
   234→                <span className="grid size-14 place-items-center rounded-full bg-emerald-100">
   235→                  <CircleCheck className="size-8 text-emerald-600" />
   236→                </span>
   237→                <div>
   238→                  <p className="text-lg font-bold text-stone-800">Pembayaran Berhasil!</p>
   239→                  <p className="mt-0.5 text-sm text-stone-500">
   240→                    {formatRupiah(payment.amount)} · {methodLabel} · {payment.invoiceNo}
   241→                  </p>
   242→                </div>
   243→                <div className="flex items-center gap-2 rounded-xl border border-emerald-200 bg-emerald-50 px-3 py-2 text-xs text-emerald-800">
   244→                  <CheckCircle2 className="size-4 shrink-0" />
   245→                  WhatsApp konfirmasi terkirim ke wali.
   246→                </div>
   247→                <Button
   248→                  onClick={onDismiss}
   249→                  className="mt-1 h-9 w-full rounded-xl bg-emerald-700 text-white hover:bg-emerald-800"
   250→                >
   251→                  Selesai
   252→                </Button>
   253→              </div>
   254→            )}
   255→          </>
   256→        )}
   257→      </DialogContent>
   258→    </Dialog>
   259→  )
   260→}

   262→// ==== QR check-in mini card ====
   263→
   264→function QrCheckinCard({
   265→  child,
   266→  activeSessions,
   267→  onRefresh,
   268→}: {
   269→  child: ParentStudent
   270→  activeSessions: SessionItem[]
   271→  onRefresh: () => void
   272→}) {
   273→  const { toast } = useToast()
   274→  const [code, setCode] = useState('')
   275→  const [checking, setChecking] = useState(false)
   276→
   277→  // Portal data has no classId, so match the child's class by class name.
   278→  const classSessions = activeSessions.filter((s) => s.className === child.className)
   279→
   280→  async function submit() {
   281→    const trimmed = code.trim()
   282→    if (!trimmed) {
   283→      toast({ title: 'Kode kosong', description: 'Masukkan kode kehadiran terlebih dahulu.', variant: 'destructive' })
   284→      return
   285→    }
   286→    setChecking(true)
   287→    try {
   288→      const res = await apiSend<{ success?: boolean; already?: boolean; message?: string }>(
   289→        '/api/attendance/checkin',
   290→        'POST',
   291→        { code: trimmed, studentId: child.id },
   292→      )
   293→      toast({ title: 'Check-in tercatat', description: res.message ?? `${child.fullName} tercatat hadir.` })
   294→      setCode('')
   295→      onRefresh()
   296→    } catch (err) {
   297→      toast({
   298→        title: 'Check-in gagal',
   299→        description: err instanceof Error ? err.message : 'Terjadi kesalahan saat check-in.',
   300→        variant: 'destructive',
   301→      })
   302→    } finally {
   303→      setChecking(false)
   304→    }
   305→  }
   306→
   307→  return (
   308→    <Card className="rounded-2xl border-emerald-200/70 lg:col-span-2">
   309→      <CardHeader>
   310→        <CardTitle className="flex items-center gap-2 text-base">
   311→          <span className="grid size-8 place-items-center rounded-lg bg-emerald-100 text-emerald-700">
   312→            <ScanLine className="size-4" />
   313→          </span>
   314→          Check-in QR Kehadiran
   315→        </CardTitle>
   316→        <CardDescription>
   317→          Masukkan kode kehadiran dari ustadz/ustadzah untuk mencatat kehadiran {child.fullName} secara mandiri.
   318→        </CardDescription>
   319→      </CardHeader>
   320→      <CardContent className="space-y-3">
   321→        <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   322→          <Input
   323→            value={code}
   324→            onChange={(e) => setCode(e.target.value)}
   325→            onKeyDown={(e) => {
   326→              if (e.key === 'Enter' && !checking) void submit()
   327→            }}
   328→            placeholder="Contoh: DJ-IA01-XXXX"
   329→            className="h-10 flex-1 rounded-xl font-mono uppercase tracking-wider"
   330→            disabled={checking}
   331→          />
   332→          <Button
   333→            onClick={() => void submit()}
   334→            disabled={checking}
   335→            className="h-10 rounded-xl bg-emerald-700 px-5 text-white hover:bg-emerald-800"
   336→          >
   337→            {checking ? (
   338→              <>
   339→                <Loader2 className="size-4 animate-spin" /> Memproses...
   340→              </>
   341→            ) : (
   342→              'Check-in'
   343→            )}
   344→          </Button>
   345→        </div>
   346→        <div className="rounded-xl bg-stone-50 p-3">
   347→          {classSessions.length > 0 ? (
   348→            <div className="flex flex-wrap items-center gap-2">
   349→              <span className="text-xs font-medium text-stone-500">Kode sesi aktif kelas {child.className}:</span>
   350→              {classSessions.map((s) => (
   351→                <button
   352→                  key={s.id}
   353→                  type="button"
   354→                  onClick={() => setCode(s.code)}
   355→                  className="rounded-full border border-amber-300 bg-amber-50 px-2.5 py-1 font-mono text-xs font-semibold text-amber-800 transition-colors hover:bg-amber-100"
   356→                  title={s.topic ? `Topik: ${s.topic}` : 'Sesi aktif'}
   357→                >
   358→                  {s.code}
   359→                </button>
   360→              ))}
   361→            </div>
   362→          ) : (
   363→            <p className="text-xs text-stone-500">
   364→              Tidak ada sesi aktif untuk kelas {child.className} saat ini. Kode muncul di sini saat ustadz/ustadzah
   365→              membuka sesi absensi.
   366→            </p>
   367→          )}
   368→        </div>
   369→      </CardContent>
   370→    </Card>
   371→  )
   372→}
