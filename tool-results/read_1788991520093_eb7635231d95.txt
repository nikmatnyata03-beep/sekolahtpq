    81→    81→function timeAgo(iso: string): string {
    82→    82→  const diff = Date.now() - new Date(iso).getTime()
    83→    83→  const m = Math.floor(diff / 60000)
    84→    84→  if (m < 1) return 'baru saja'
    85→    85→  if (m < 60) return `${m} menit lalu`
    86→    86→  const h = Math.floor(m / 60)
    87→    87→  if (h < 24) return `${h} jam lalu`
    88→    88→  const d = Math.floor(h / 24)
    89→    89→  if (d < 7) return `${d} hari lalu`
    90→    90→  return formatShortDate(iso)
    91→    91→}

   389→   389→export function GuruOverview({ user, onNavigate }: {
   390→   390→  user: AuthUser
   391→   391→  onNavigate?: (section: GuruOverviewSection) => void
   392→   392→}) {
   393→   393→  const { toast } = useToast()
   394→   394→  const mountedRef = useRef(true)
   395→   395→  const [data, setData] = useState<GuruData | null>(null)
   396→   396→  const [loading, setLoading] = useState(true)
   397→   397→  const [error, setError] = useState<string | null>(null)
   398→   398→  // Quick Setoran (Task 15-a): id santri yang sheet-nya terbuka (null = tertutup).
   399→   399→  const [quickStudentId, setQuickStudentId] = useState<string | null>(null)
   400→   400→
   401→   401→  // quiet=true (Task 15-a): segarkan data TANPA skeleton/error penuh — dipakai
   402→   402→  // setelah simpan setoran cepat agar bar kartu Progres Target langsung update.
   403→   403→  const load = useCallback(async (quiet = false) => {
   404→   404→    if (!quiet) {
   405→   405→      setLoading(true)
   406→   406→      setError(null)
   407→   407→    }
   408→   408→    try {
   409→   409→      const [classes, sessions, hafalans, students] = await Promise.all([
   410→   410→        apiGet<ClassRoom[]>('/api/classes'),
   411→   411→        apiGet<SessionItem[]>('/api/sessions?active=1'),
   412→   412→        apiGet<Hafalan[]>('/api/hafalan'),
   413→   413→        apiGet<Student[]>('/api/students'),
   414→   414→      ])
   415→   415→      if (!mountedRef.current) return
   416→   416→      setData({ classes, sessions, hafalans, students })
   417→   417→    } catch (e) {
   418→   418→      if (!mountedRef.current) return
   419→   419→      if (quiet) return // refresh senyap: biarkan data lama tampil tanpa Alert penuh
   420→   420→      setError(e instanceof Error ? e.message : 'Gagal memuat data ringkasan')
